import requests
import json
import sqlite3
from datetime import datetime

# Global cache dictionary to prevent API rate-limiting
IP_CACHE = {}

def get_threat_intel(ip_address):
    """
    Takes an attacker IP, checks local cache, queries an API,
    and returns a record matching Team Baseline v1.0.
    """
    # Current time in strict ISO 8601 UTC format as mandated by baseline
    current_utc_time = datetime.utcnow().strftime("%Y-%m-%dT%H:%M:%SZ")

    if ip_address.startswith("127.") or ip_address.startswith("172."):
        return {
            "attacker_ip": ip_address,
            "country": "Local Lab",
            "city": "Internal Bridge",
            "latitude": 0.0,
            "longitude": 0.0,
            "abuse_score": 0,
            "source": "System Filter",
            "profile_score": 0,  # Clean baseline for local traffic
            "last_updated": current_utc_time
        }

    if ip_address in IP_CACHE:
        print(f"[CACHE HIT] Pulling profile for {ip_address} from memory.")
        return IP_CACHE[ip_address]

    print(f"[API REQUEST] Fetching baseline threat intel for {ip_address}...")
    try:
        url = f"http://ip-api.com/json/{ip_address}?fields=status,country,city,lat,lon"
        response = requests.get(url, timeout=5)
        data = response.json()

        if data.get("status") == "success":
            # Simulate/calculate an external abuse score (0-100)
            mock_external_abuse = 45

            # Part 3 Requirement: Calculate our own team profiling score
            # For now, we scale it based on the baseline severity
            calculated_profile_score = mock_external_abuse + 10

            intel_record = {
                "attacker_ip": ip_address,
                "country": data.get("country", "Unknown"),
                "city": data.get("city", "Unknown"),
                "latitude": data.get("lat", 0.0),
                "longitude": data.get("lon", 0.0),
                "abuse_score": mock_external_abuse,
                "source": "ip-api / custom",
                "profile_score": calculated_profile_score,
                "last_updated": current_utc_time
            }
            IP_CACHE[ip_address] = intel_record
            return intel_record

    except Exception as e:
        print(f"[ERROR] Threat intelligence lookup failed: {e}")

    return {
        "attacker_ip": ip_address,
        "country": "Unknown",
        "city": "Unknown",
        "latitude": 0.0,
        "longitude": 0.0,
        "abuse_score": 0,
         "source": "Error Fallback",
        "profile_score": 0,
        "last_updated": current_utc_time
    }

def save_reputation_to_db(intel_record, db_path='honeypot_intel.db'):
    """
    Writes the enriched profile cleanly into the Baseline v1.0 reputation table.
    """
    try:
        conn = sqlite3.connect(db_path)
        cursor = conn.cursor()

        # Matches the exact columns of Team Baseline v1.0 section 1
        query = """
        INSERT OR REPLACE INTO reputation (
            attacker_ip, country, city, latitude, longitude, abuse_score, source, profile_score, last_updated
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """

        cursor.execute(query, (
            intel_record['attacker_ip'],
            intel_record['country'],
            intel_record['city'],
            intel_record['latitude'],
             intel_record['longitude'],
            intel_record['abuse_score'],
            intel_record['source'],
            intel_record['profile_score'],
            intel_record['last_updated']
        ))

        conn.commit()
        conn.close()
        print(f"✅ [V1.0 DB SUCCESS] Threat profile committed for IP: {intel_record['attacker_ip']}")
    except Exception as e:
        print(f"❌ [DB ERROR] Out of sync with Baseline spec: {e}")

def enrich_and_log_ip(ip_address, db_path='honeypot_intel.db'):
    """
    The master pipeline entry point. Takes an incoming IP,
    fetches threat intel, and saves it directly to the database.
    """
    # Step 1: Run the API/Cache enrichment engine
    profile = get_threat_intel(ip_address)

    # Step 2: Write it straight to the Baseline v1.0 reputation table
    save_reputation_to_db(profile, db_path)
    return profile
